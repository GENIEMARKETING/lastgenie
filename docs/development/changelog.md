- **Action: Generate Initial Prisma Schema.**
  - **Prompt:** "Generate a Prisma schema for a PostgreSQL database. Include models for `User`, `Product`, and a `Subscription`. The `Product` model should have fields for name, description, price, and sku. The `User` model needs basic auth fields. The `Subscription` model should link a `User` to a `Product` and include a status and renewal date."
  - **Result:** The AI produced a robust starting schema, which serves as the foundation for the database.

    ```prisma
    // prisma/schema.prisma

    generator client {
      provider = "prisma-client-js"
    }

    datasource db {
      provider = "postgresql"
      url      = env("DATABASE_URL")
    }

    model User {
      id            String    @id @default(cuid())
      email         String    @unique
      passwordHash  String
      createdAt     DateTime  @default(now())
      updatedAt     DateTime  @updatedAt
      subscriptions Subscription[]
      orders        Order[]
    }

    model Product {
      id            String    @id @default(cuid())
      sku           String    @unique
      name          String
      description   String
      price         Int // Store price in cents to avoid floating point issues
      isPackage     Boolean   @default(false)
      createdAt     DateTime  @default(now())
      subscriptions Subscription[]
      orderItems    OrderItem[]
    }

    model Subscription {
      id        String   @id @default(cuid())
      userId    String
      user      User     @relation(fields: [userId], references: [id])
      productId String
      product   Product  @relation(fields: [productId], references: [id])
      status    String // e.g., "active", "paused", "cancelled"
      nextRenewalDate DateTime
      createdAt DateTime @default(now())
      updatedAt DateTime @updatedAt
    }
    
    // Additional models for one-time purchases
    model Order {
      id        String      @id @default(cuid())
      userId    String
      user      User        @relation(fields: [userId], references: [id])
      total     Int
      status    String      // e.g., "pending", "shipped", "delivered"
      createdAt DateTime    @default(now())
      items     OrderItem[]
    }

    model OrderItem {
      id        String  @id @default(cuid())
      orderId   String
      order     Order   @relation(fields: [orderId], references: [id])
      productId String
      product   Product @relation(fields: [productId], references: [id])
      quantity  Int
      price     Int
    }
    ```
